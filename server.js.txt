require('dotenv').config();
const express = require('express');
const http = require('http');
const { Server } = require('socket.io');
const { WebcastPushConnection } = require('tiktok-live-connector');
const path = require('path');
const cors = require('cors');

const app = express();
const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: '*' }
});

app.use(cors());
app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

const activeConnections = new Map();

// Endpoint TTS tiếng Việt (hỗ trợ phát mượt trên máy tính & iPhone)
app.get('/api/tts', async (req, res) => {
  try {
    const text = (req.query.text || '').toString().trim();
    if (!text) return res.status(400).send('Thiếu nội dung văn bản');

    const chunks = [];
    const maxLen = 180;
    for (let i = 0; i < text.length; i += maxLen) {
      chunks.push(text.slice(i, i + maxLen));
    }

    const audioBuffers = [];
    for (const part of chunks) {
      const url = `https://translate.google.com/translate_tts?ie=UTF-8&q=${encodeURIComponent(part)}&tl=vi&client=tw-ob`;
      const response = await fetch(url, {
        headers: {
          'User-Agent': 'Mozilla/5.0 (iPhone; CPU iPhone OS 17_0 like Mac OS X) AppleWebKit/605.1.15'
        }
      });
      if (!response.ok) throw new Error('Lỗi tải âm thanh TTS');
      const arrBuf = await response.arrayBuffer();
      audioBuffers.push(Buffer.from(arrBuf));
    }

    res.set({
      'Content-Type': 'audio/mpeg',
      'Cache-Control': 'no-cache'
    });
    res.send(Buffer.concat(audioBuffers));
  } catch (err) {
    console.error('[TTS Error]:', err.message);
    res.status(500).send('Không thể tạo giọng nói');
  }
});

// Endpoint gọi Gemini API nếu người dùng cấu hình GEMINI_API_KEY
app.post('/api/ai-comment', async (req, res) => {
  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) return res.json({ useLocalEngine: true });

  const { eventType, userName, giftName, giftCount, commentText } = req.body;
  let prompt =
    'Bạn là MC livestream TikTok theo phong cách tu tiên, tiên hiệp cổ trang. Hãy viết 1 câu tiếng Việt tự nhiên, ngắn gọn (dưới 25 từ), dùng từ như đạo hữu, tu vi, cảnh giới, tông môn, linh lực, độ kiếp, lão tổ, tiên tôn, phi thăng. Không dùng emoji. ';

  if (eventType === 'join') prompt += `Người xem "${userName}" vừa vào phòng live. Chào mừng họ.`;
  else if (eventType === 'gift') prompt += `Đạo hữu "${userName}" vừa tặng ${giftCount}x "${giftName}". Cảm ơn nồng nhiệt.`;
  else if (eventType === 'chat') prompt += `Đạo hữu "${userName}" bình luận: "${commentText}". Đáp lại hay ho.`;

  try {
    const r = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`,
      {
        method: 'POST',
headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
      }
    );
    const data = await r.json();
    const text = data?.candidates?.[0]?.content?.parts?.[0]?.text?.trim();
    if (text) return res.json({ text, useLocalEngine: false });
    res.json({ useLocalEngine: true });
  } catch {
    res.json({ useLocalEngine: true });
  }
});

io.on('connection', (socket) => {
  socket.on('connect-tiktok', async (username) => {
    if (activeConnections.has(socket.id)) {
      activeConnections.get(socket.id).disconnect();
      activeConnections.delete(socket.id);
    }

    const cleanUser = (username || '').replace(/^@/, '').trim();
    if (!cleanUser) {
      socket.emit('tiktok-error', 'Vui lòng nhập ID TikTok hợp lệ.');
      return;
    }

    socket.emit('tiktok-status', {
      status: 'connecting',
      message: `Đang kết nối tới phòng LIVE @${cleanUser}...`
    });

    const conn = new WebcastPushConnection(cleanUser, {
      processInitialData: false,
      enableExtendedGiftInfo: true,
      enableWebsocketUpgrade: true,
      requestPollingIntervalMs: 2000
    });

    try {
      const state = await conn.connect();
      activeConnections.set(socket.id, conn);

      socket.emit('tiktok-status', {
        status: 'connected',
        message: `Đã kết nối LIVE @${cleanUser} (Room: ${state.roomId})`
      });

      conn.on('member', (data) => {
        socket.emit('live-event', {
          type: 'join',
          nickname: data.nickname || data.uniqueId,
          uniqueId: data.uniqueId,
          timestamp: Date.now()
        });
      });

      conn.on('chat', (data) => {
        socket.emit('live-event', {
          type: 'chat',
          nickname: data.nickname || data.uniqueId,
          uniqueId: data.uniqueId,
          comment: data.comment,
          timestamp: Date.now()
        });
      });

      conn.on('gift', (data) => {
        if (data.giftType === 1 && !data.repeatEnd) return;
        const repeatCount = data.repeatCount || 1;
        const diamondValue = (data.diamondCount || 1) * repeatCount;
        socket.emit('live-event', {
          type: 'gift',
          nickname: data.nickname || data.uniqueId,
          uniqueId: data.uniqueId,
          giftName: data.giftName || 'Linh Bảo',
          giftCount: repeatCount,
          diamondValue,
          timestamp: Date.now()
        });
      });

      conn.on('streamEnd', () => {
        socket.emit('tiktok-status', {
          status: 'disconnected',
          message: 'Phiên livestream đã kết thúc.'
        });
      });

      conn.on('disconnected', () => {
        socket.emit('tiktok-status', {
          status: 'disconnected',
          message: 'Đã ngắt kết nối khỏi TikTok LIVE.'
        });
      });
    } catch (err) {
      socket.emit(
        'tiktok-error',
`Không thể kết nối @${cleanUser}. Kiểm tra xem kênh có đang phát LIVE công khai không. (${err.message})`
      );
    }
  });

  socket.on('disconnect-tiktok', () => {
    if (activeConnections.has(socket.id)) {
      activeConnections.get(socket.id).disconnect();
      activeConnections.delete(socket.id);
    }
    socket.emit('tiktok-status', {
      status: 'disconnected',
      message: 'Đã ngắt kết nối.'
    });
  });

  socket.on('disconnect', () => {
    if (activeConnections.has(socket.id)) {
      activeConnections.get(socket.id).disconnect();
      activeConnections.delete(socket.id);
    }
  });
});

const PORT = process.env.PORT || 3000;
server.listen(PORT, () => {
  console.log(`TikTok LIVE AI MC đang chạy tại http://localhost:${PORT}`);
});
