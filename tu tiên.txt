const express = require('express'); const http = require('http'); const { Server } = require('socket.io'); const { WebcastPushConnection } = require('tiktok-live-connector'); const path = require('path'); const cors = require('cors'); const app = express(); const server = http.createServer(app); const io = new Server(server, { cors: { origin: "*" } }); app.use(cors()); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // Lưu trữ kết nối TikTok theo từng phiên client const activeConnections = new Map(); // API Endpoint hỗ trợ đọc TTS tiếng Việt qua Google Translate TTS (Giọng nữ Bắc chuẩn, không bị giới hạn thiết bị trên iPhone) app.get('/api/tts', async (req, res) => { try { const text = req.query.text; const lang = req.query.lang || 'vi'; if (!text) return res.status(400).send('Thiếu nội dung văn bản'); // Chia nhỏ văn bản nếu quá 200 ký tự để đảm bảo API hoạt động ổn định const encodedText = encodeURIComponent(text.substring(0, 200)); const url = `https://translate.google.com/translate_tts?ie=UTF-8&q=${encodedText}&tl=${lang}&client=tw-ob`; const response = await fetch(url, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36' } }); if (!response.ok) throw new Error('Không thể tải luồng âm thanh'); res.set({ 'Content-Type': 'audio/mpeg', 'Cache-Control': 'no-cache' }); const arrayBuffer = await response.arrayBuffer(); res.send(Buffer.from(arrayBuffer)); } catch (error) { console.error('[TTS Proxy Error]:', error.message); res.status(500).send('Lỗi tạo âm thanh'); } }); // API Endpoint sinh câu thoại Tu Tiên bằng AI (Nếu có cấu hình GEMINI_API_KEY) app.post('/api/ai-comment', async (req, res) => { const { eventType, userName, giftName, giftCount, commentText } = req.body; const apiKey = process.env.GEMINI_API_KEY; if (!apiKey) { return res.json({ useLocalEngine: true }); } try { let prompt = `Bạn là một AI MC dẫn chương trình livestream TikTok theo phong cách tiên hiệp, tu tiên cổ trang. Hãy viết 1 câu bình luận ngắn gọn (dưới 25 từ), tự nhiên, giàu khí chất tu tiên bằng tiếng Việt, dùng các từ như đạo hữu, tu vi, cảnh giới, tông môn, linh lực, độ kiếp. Không dùng emoji hay ký tự đặc biệt. `; if (eventType === 'join') { prompt += `Người xem tên "${userName}" vừa vào phòng live. Hãy chào mừng họ.`; } else if (eventType === 'gift') { prompt += `Đạo hữu "${userName}" vừa tặng ${giftCount}x "${giftName}". Hãy cảm ơn nồng nhiệt.`; } else if (eventType === 'chat') { prompt += `Đạo hữu "${userName}" vừa bình luận: "${commentText}". Hãy đáp lại thú vị.`; } const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await response.json(); const aiText = data?.candidates?.[0]?.content?.parts?.[0]?.text?.trim(); if (aiText) { return res.json({ text: aiText, useLocalEngine: false }); } res.json({ useLocalEngine: true }); } catch (err) { res.json({ useLocalEngine: true }); } }); // Xử lý kết nối WebSocket thời gian thực io.on('connection', (socket) => { console.log(`[Client Connected]: ${socket.id}`); // Lệnh kết nối tới TikTok LIVE chính thức (qua UniqueId công khai) socket.on('connect-tiktok', async (username) => { // Ngắt kết nối cũ nếu đang tồn tại if (activeConnections.has(socket.id)) { const oldConn = activeConnections.get(socket.id); oldConn.disconnect(); activeConnections.delete(socket.id); } const cleanUsername = username.replace('@', '').trim(); if (!cleanUsername) { socket.emit('tiktok-error', 'Vui lòng nhập TikTok ID (ví dụ: @ten_kenh)'); return; } socket.emit('tiktok-status', { status: 'connecting', message: `Đang kết nối linh thức tới @${cleanUsername}...` }); const tiktokConnection = new WebcastPushConnection(cleanUsername, { processInitialData: false, enableExtendedGiftInfo: true, enableWebsocketUpgrade: true, requestPollingIntervalMs: 2000 }); try { const state = await tiktokConnection.connect(); activeConnections.set(socket.id, tiktokConnection); console.log(`[TikTok Connected]: @${cleanUsername} (RoomID: ${state.roomId})`); socket.emit('tiktok-status', { status: 'connected', message: `Đã kết nối thành công với Đạo Trường @${cleanUsername}`, roomId: state.roomId }); // 1. Sự kiện người xem mới vào phòng tiktokConnection.on('member', (data) => { socket.emit('live-event', { type: 'join', userId: data.userId, nickname: data.nickname || data.uniqueId, uniqueId: data.uniqueId, timestamp: Date.now() }); }); // 2. Sự kiện bình luận tiktokConnection.on('chat', (data) => { socket.emit('live-event', { type: 'chat', userId: data.userId, nickname: data.nickname || data.uniqueId, uniqueId: data.uniqueId, comment: data.comment, timestamp: Date.now() }); }); // 3. Sự kiện tặng quà tiktokConnection.on('gift', (data) => { // Chỉ xử lý khi chuỗi combo kết thúc hoặc quà tặng đơn lẻ (tránh đọc lặp 100 lần khi user bấm liên tục) if (data.giftType === 1 && !data.repeatEnd) { return; } const repeatCount = data.repeatCount || 1; const diamondCount = (data.diamondCount || 1) * repeatCount; socket.emit('live-event', { type: 'gift', userId: data.userId, nickname: data.nickname || data.uniqueId, uniqueId: data.uniqueId, giftName: data.giftName || 'Linh Bảo', giftCount: repeatCount, diamondValue: diamondCount, timestamp: Date.now() }); }); // Sự kiện kết thúc live hoặc mất kết nối tiktokConnection.on('streamEnd', () => { socket.emit('tiktok-status', { status: 'disconnected', message: 'Phiên livestream đã kết thúc.' }); }); tiktokConnection.on('disconnected', () => { socket.emit('tiktok-status', { status: 'disconnected', message: 'Đã ngắt kết nối khỏi TikTok LIVE.' }); }); } catch (err) { console.error(`[TikTok Connect Failed]:`, err.message); socket.emit('tiktok-error', `Không thể kết nối tới @${cleanUsername}. Đảm bảo kênh đang phát LIVE công khai! (${err.message})`); } }); // Ngắt kết nối thủ công socket.on('disconnect-tiktok', () => { if (activeConnections.has(socket.id)) { activeConnections.get(socket.id).disconnect(); activeConnections.delete(socket.id); socket.emit('tiktok-status', { status: 'disconnected', message: 'Đã chủ động ngắt kết nối.' }); } }); // Dọn dẹp khi đóng trình duyệt socket.on('disconnect', () => { if (activeConnections.has(socket.id)) { activeConnections.get(socket.id).disconnect(); activeConnections.delete(socket.id); } console.log(`[Client Disconnected]: ${socket.id}`); }); }); const PORT = process.env.PORT || 3000; server.listen(PORT, () => { console.log(`====================================================`); console.log(`🚀 TIKTOK LIVE AI MC (TU TIÊN EDITION) ĐANG CHẠY!`); console.log(`🌐 Truy cập trên máy tính: http://localhost:${PORT}`); console.log(`📱 Truy cập trên iPhone: http://<IP-MAY-TINH>:${PORT}`); console.log(`====================================================`); });
